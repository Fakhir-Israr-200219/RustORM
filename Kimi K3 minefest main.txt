Ye wo **generic engine** hai jo crates.io pe hoga:

```rust
// rustorm/src/lib.rs (published crate)
pub mod database;
pub mod query;
pub mod field;
pub mod error;

// Core trait - har generated entity ye implement karegi
pub trait Entity: Sized + Send + Sync + 'static {
    const TABLE_NAME: &'static str;
    type Model: Send + Sync + Unpin + for<'r> sqlx::FromRow<'r, sqlx::postgres::PgRow>;
}

// Generic Field
pub struct Field<T> {
    pub name: &'static str,
    _marker: std::marker::PhantomData<T>,
}

impl<T> Field<T> {
    pub const fn new(name: &'static str) -> Self {
        Self { name, _marker: std::marker::PhantomData }
    }
    pub fn eq(&self, value: T) -> Condition {
        Condition::Eq(self.name, /* serialized value */)
    }
}

// Query builder
pub struct Query<E: Entity> {
    pub table: &'static str,
    pub conditions: Vec<Condition>,
    pub limit: Option<u64>,
    pub order_by: Vec<Order>,
    _marker: std::marker::PhantomData<E>,
}

impl<E: Entity> Query<E> {
    pub fn new() -> Self { /* ... */ }
    pub fn where_(mut self, cond: Condition) -> Self { /* ... */ self }
    pub fn take(mut self, n: u64) -> Self { self.limit = Some(n); self }
    
    pub async fn all(self, db: &Database) -> Result<Vec<E::Model>, RustORMError> {
        // 1. SQL generate karo
        // 2. SQLx se execute karo
        // 3. Vec<E::Model> return karo
    }
}

// Database wrapper
pub struct Database {
    pub(crate) pool: sqlx::PgPool, // ya generic pool
}

impl Database {
    pub async fn connect(url: &str) -> Result<Self, RustORMError> {
        let pool = sqlx::postgres::PgPool::connect(url).await?;
        Ok(Self { pool })
    }
}
```

## 5. User ka actual code (Axum app)

```rust
// src/main.rs
use axum::{
    routing::get,
    Router,
    extract::State,
    Json,
};
use rustorm::Database;
use rustorm_entities::{User, UserModel};  // ← Generated se
use std::sync::Arc;

#[derive(Clone)]
struct AppState {
    db: Database,
}

#[tokio::main]
async fn main() {
    let db = Database::connect("postgres://localhost/mydb").await.unwrap();
    let state = AppState { db };
    
    let app = Router::new()
        .route("/users", get(get_users))
        .with_state(state);
        
    let listener = tokio::net::TcpListener::bind("0.0.0.0:3000").await.unwrap();
    axum::serve(listener, app).await.unwrap();
}

async fn get_users(
    State(state): State<AppState>,
) -> Result<Json<Vec<UserModel>>, String> {
    let users = User::find()           // ← User::find() generated hai
        .where_(User::name.eq("Fakhir".to_string()))  // ← Type-safe!
        .take(20)
        .all(&state.db)                // ← Database connection yahan pass hoti hai
        .await
        .map_err(|e| e.to_string())?;
        
    Ok(Json(users))
}
```
